# Voca - local-first push-to-talk dictation for macOS (Apple Silicon)
# Locked stack per spec.md. Pinned to major versions known to work on M-series / Python 3.12.

# On-device speech-to-text on the Apple GPU (Metal) via MLX
mlx-whisper>=0.4.0

# Audio capture: 16kHz mono float32 from the mic
sounddevice>=0.4.6
numpy>=1.26,<2.0
soundfile>=0.12.1

# Global hotkey capture + key injection use the native macOS Quartz APIs
# (CGEventTap / CGEvent via pyobjc, pulled in by rumps). We do NOT use pynput:
# it crashes inside a bundled .app by calling the Text Input Source API off the
# main thread.

# Clipboard read/write
pyperclip>=1.8.2

# macOS menu bar app
rumps>=0.4.0

# macOS frameworks the app imports directly. rumps only pulls Cocoa, so these
# must be listed explicitly or a fresh install breaks:
#   Quartz              -> CGEventTap (global hotkey) + CGEvent (paste)
#   ApplicationServices -> Accessibility (AXUIElement, AXIsProcessTrusted)
pyobjc-framework-Quartz>=10
pyobjc-framework-ApplicationServices>=10

# Local English punctuation restoration (felflare/bert-restore-punctuation).
# Runs on CPU/MPS via torch (pulled in transitively); used only during a
# dictation, never at idle.
transformers>=4.40,<5

# Voice-activity detection to gate out silence/noise before transcription,
# so Whisper doesn't hallucinate text. Tiny on-device model; runs on torch.
# Pinned to 6.2.1: 6.2.2 hard-imports onnxruntime at package load, which we
# don't bundle (we use the torch JIT model), breaking the app bundle.
silero-vad==6.2.1

# Optional AI cleanup layer (enhance.py). Local backend: a small on-device LLM
# on the Apple GPU (free, offline) that fixes grammar, drops fillers, and
# smart-formats numbers/dates/emails/lists. Cloud backend: Anthropic Claude via
# the user's own API key (opt-in). Both degrade to raw text if unavailable.
mlx-lm>=0.31,<1
anthropic>=0.40
